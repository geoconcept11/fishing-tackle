function evaluateCoherence(pMin, pMax, peValue, fluoroDiam, lureWeight, lureType) { let alerts = []; let score = 10; // Note sur 10 // 1. Test Masse vs Canne if (lureWeight < pMin) { alerts.push({ type: 'red', text: `Leurre de ${lureWeight}g trop léger pour la canne (${pMin}-${pMax}g). Perte de distance et de sensations.` }); score -= 4; } else if (lureWeight > pMax) { alerts.push({ type: 'red', text: `DANGER : Leurre de ${lureWeight}g supérieur à la limite max de la canne (${pMax}g). Risque de casse au lancer !` }); score -= 5; } // 2. Test Tresse vs Canne let targetPE = pMax / 25; if (peValue > targetPE + 1.0) { alerts.push({ type: 'orange', text: `Tresse (PE ${peValue}) surdimensionnée par rapport à la canne. Perte de sensibilité.` }); score -= 2; } else if (peValue < targetPE - 0.5) { alerts.push({ type: 'orange', text: `Tresse (PE ${peValue}) un peu fine pour la puissance de cette canne.` }); score -= 2; } // 3. Test Rigidité Bas de Ligne if (lureWeight <= 5 && fluoroDiam > 30) { alerts.push({ type: 'orange', text: `Fluorocarbone de ${fluoroDiam}/100 trop épais pour un leurre de ${lureWeight}g : la nage va être bridée.` }); score -= 2; } // 4. Test Sécurité Dentition Brochet if ((lureWeight > 35 || ['JK', 'GB', 'SB'].includes(lureType)) && fluoroDiam < 60) { alerts.push({ type: 'red', text: `Attention : Bas de ligne de ${fluoroDiam}/100 insuffisant face aux dents de brochet. Risque de coupe !` }); score -= 3; } if (score < 0) score = 0; return { score, alerts }; }
